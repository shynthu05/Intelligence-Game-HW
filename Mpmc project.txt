#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>
// OLED settings
#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, -1);
// LED pins
const int leds[4] = {13, 12, 11, 10};
// Button pins: Red, Green, Blue, Yellow
const int buttons[4] = {8, 7, 6, 5};
const int startButton = 9;
const int buzzer = 3;
const char* colours[4] = {
"RED", "GREEN", "BLUE", "YELLOW"
};
// Four level time limits in milliseconds
const unsigned long timeLimits[4] = {
2000, 1500, 1000, 500
};
int level = 0;
int score = 0;
int lives = 3;
int rounds = 0;
int target = 0;
unsigned long startTime = 0;
unsigned long reactionTime = 0;
bool gameRunning = false;
bool waitingForStart = true;
bool finished = false;
void allLedsOff() {
for (int i = 0; i < 4; i++) {
digitalWrite(leds[i], LOW);
}
}
void showMessage(const char* line1, const char* line2) {
display.clearDisplay();
display.setTextSize(1);
display.setTextColor(SSD1306_WHITE);
display.setCursor(0, 15);
display.println(line1);
display.println();
display.println(line2);
display.display();
}
void showStatus() {
display.clearDisplay();
display.setTextSize(1);
display.setTextColor(SSD1306_WHITE);
display.setCursor(0, 0);
display.println("LED REACTION GAME");
display.print("Level: ");
display.println(level + 1);
display.print("Score: ");
display.println(score);
display.print("Lives: ");
display.println(lives);
display.print("Time limit: ");
display.print(timeLimits[level]);
display.println(" ms");
display.display();
}
void feedback(bool correct) {
if (correct) {
tone(buzzer, 2000, 120);
} else {
tone(buzzer, 400, 300);
}
delay(350);
}
void waitForRelease(int pin) {
while (digitalRead(pin) == LOW) {
delay(10);
}
delay(30);
}
void startGame() {
level = 0;
score = 0;
lives = 3;
rounds = 0;
finished = false;
gameRunning = true;
waitingForStart = false;
showStatus();
delay(700);
}
void finishGame(bool won) {
gameRunning = false;
finished = true;
allLedsOff();
if (won) {
showMessage("CONGRATULATIONS!", "ALL LEVELS DONE!");
tone(buzzer, 1800, 400);
} else {
showMessage("GAME OVER", "Try again!");
tone(buzzer, 300, 500);
}
delay(1500);
display.clearDisplay();
display.setCursor(0, 5);
display.setTextSize(1);
display.setTextColor(SSD1306_WHITE);
display.println("FINAL RESULT");
display.print("Score: ");
display.println(score);
display.print("Level reached: ");
display.println(level + 1);
display.display();
}
void setup() {
Serial.begin(9600);
for (int i = 0; i < 4; i++) {
pinMode(leds[i], OUTPUT);
pinMode(buttons[i], INPUT_PULLUP);
}
pinMode(startButton, INPUT_PULLUP);
pinMode(buzzer, OUTPUT);
allLedsOff();
if (!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
Serial.println("OLED not detected!");
while (true) {
// Check OLED wiring and I2C address
}
}
randomSeed(analogRead(A0));
showMessage("LED REACTION GAME", "Press START");
}
void loop() {
// Wait for Start button
if (!gameRunning) {
if (digitalRead(startButton) == LOW) {
delay(30);
if (digitalRead(startButton) == LOW) {
waitForRelease(startButton);
startGame();
}
}
return;
}
showStatus();
// Select a random LED
allLedsOff();
target = random(0, 4);
digitalWrite(leds[target], HIGH);
// Start reaction timer
startTime = millis();
bool answered = false;
while (!answered &&
millis() - startTime < timeLimits[level]) {
for (int i = 0; i < 4; i++) {
if (digitalRead(buttons[i]) == LOW) {
delay(25); // Simple button debounce
if (digitalRead(buttons[i]) == LOW) {
reactionTime = millis() - startTime;
waitForRelease(buttons[i]);
allLedsOff();
answered = true;
tone(buzzer, 1500, 200);
delay(700);
} else {
finishGame(true);
return;
}
}
} else {
lives--;
Serial.println("Wrong button!");
showMessage("WRONG BUTTON!", "Life lost");
feedback(false);
if (lives <= 0) {
finishGame(false);
return;
}
}
break;
}
}
}
}
// No button pressed before deadline
if (!answered) {
allLedsOff();
lives--;
showMessage("TIME OUT!", "Life lost");
feedback(false);
if (lives <= 0) {
finishGame(false);
return;
}
}
showStatus();
delay(400);
}
